cmake_minimum_required(VERSION 3.21)
project(Mallard LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OSX_ARCHITECTURES arm64)
set(PROJECT_ARCH arm64)
set(USE_SANDBOX ON CACHE BOOL "Keep Chromium helper processes sandboxed")
set(CEF_ROOT "${CMAKE_SOURCE_DIR}/.deps/cef")
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0")
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)

add_executable(Mallard MACOSX_BUNDLE Sources/App.mm Sources/Chrome.mm Sources/Favicons.mm Sources/Telemetry.mm Sources/Bookmarks.mm Sources/BookmarkView.mm Sources/Core.mm Sources/Vault.mm Sources/FilterUpdater.mm)
target_include_directories(Mallard PRIVATE "${CEF_ROOT}" Sources)
target_compile_definitions(Mallard PRIVATE ${CEF_COMPILER_DEFINES})
target_compile_options(Mallard PRIVATE -fobjc-arc -Wno-deprecated-declarations)
target_link_libraries(Mallard PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS} "-framework Cocoa" "-framework Security" "-framework LocalAuthentication")
target_link_libraries(Mallard PRIVATE "-framework ImageIO" sqlite3)
target_link_libraries(Mallard PRIVATE "${CMAKE_SOURCE_DIR}/Blocker/target/release/libmallard_blocker.a" "-framework SystemConfiguration")
set_target_properties(Mallard PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/Resources/Info.plist")

add_executable(MallardHelper Sources/Helper.mm)
target_include_directories(MallardHelper PRIVATE "${CEF_ROOT}" Sources)
target_compile_definitions(MallardHelper PRIVATE ${CEF_COMPILER_DEFINES})
target_compile_options(MallardHelper PRIVATE -fobjc-arc)
target_link_libraries(MallardHelper PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS} "-framework Cocoa")
target_link_libraries(MallardHelper PRIVATE "${CMAKE_SOURCE_DIR}/Blocker/target/release/libmallard_blocker.a" "-framework Security" "-framework SystemConfiguration")
